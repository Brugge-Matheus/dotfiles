waybar            # barra de status
awww              # wallpaper com transicoes (= swww renomeado; Provides/Replaces swww)
swaync            # notificacoes + central
hyprlock          # lockscreen (oficial Hypr)
hypridle          # idle daemon (oficial Hypr)
brightnessctl     # teclas de brilho
playerctl         # teclas de midia (play/pause/next)
wl-clipboard      # copiar/colar no Wayland (wl-copy/wl-paste)
libnotify         # notify-send (enviar notificacoes via CLI/scripts)
gsimplecal        # popup de calendario (clique no relogio da Waybar)
fuzzel            # menu dmenu minimalista (power menu, wifi, settings)
rofi              # launcher principal (SUPER+Espaco). rofi 2.0+ = Wayland nativo
libqalculate      # motor da calculadora dev (o `qalc`, usado pelo rofi-calc-dev.sh, SUPER+=)
rofi-calc         # calculadora "live" alternativa (plugin); tambem puxa a libqalculate
translate-shell   # extensao: tradutor (o `trans`, usado pelo rofi-translate.sh, SUPER+T)
power-profiles-daemon  # perfis de energia (controle pelo botao da bateria)
upower            # estado/tempo da bateria (tooltip do modulo custom/battery)

yt-dlp            # deixa o mpv tocar links do YouTube/etc direto (mpv <url>)
